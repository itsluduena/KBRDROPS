function doGet(e) { if (e && e.parameter && e.parameter.api === 'obtenerProductos') { return obtenerProductosJSON(); } return HtmlService.createTemplateFromFile('index') .evaluate() .setTitle('KBRDROPS — Catálogo Oficial') .addMetaTag('viewport', 'width=device-width, initial-scale=1.0, maximum-scale=1.0, user-scalable=no') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL); } function obtenerProductosJSON() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var datos = sheet.getDataRange().getValues(); var productos = []; for (var i = 1; i < datos.length; i++) { var fila = datos[i]; if (!fila[0] && !fila[1]) continue; // Captura dinámica de imágenes filtrando vacíos var fotos = []; for (var j = 4; j < fila.length - 1; j++) { var val = String(fila[j]).trim(); if (val !== '' && val.startsWith('http')) { fotos.push(val); } } // La última columna siempre corresponde al modelo 3D var modelo3d = String(fila[fila.length - 1]).trim(); productos.push({ id: fila[0], nombre: fila[1], precio: fila[2], descripcion: fila[3], fotos: fotos, modelo3d: modelo3d.startsWith('http') ? modelo3d : '' }); } return ContentService.createTextOutput(JSON.stringify(productos)) .setMimeType(ContentService.MimeType.JSON); }
